36class Simplex2D: 

37 top_corner = np.sqrt(3) / 2 

38 side_slope = np.sqrt(3) 

39 corners = np.array([ 

40 [0, 0], 

41 [1 / 2, top_corner], 

42 [1, 0] 

43 ]) 

44 triangle = tri.Triangulation(corners[:, 0], corners[:, 1]) 

45 refiner = tri.UniformTriRefiner(triangle) 

46 trimesh = refiner.refine_triangulation(subdiv=5)

48 def __init__(self, nb_points: Optional[int] = 1000, discrete: Optional[bool] = False, 

49 size: Optional[Union[int, None]] = None): 

50 """ 

51 Plots a 2-dimensional simplex in a cartesian plane. 

52 

53 This class offers utility methods to plot gradients and equilibrium points on a 2-simplex (triangle). 

54 

55 The plotting is always done on the unit simplex for convenience. At the moment no rotations are 

56 implemented, but we plan to add this feature, so that the triangle can be rotated before the plot. 

57 

58 We discern between continuous and discrete dynamics. The main reason is that this class' objective 

59 is to plot evolutionary dynamics on a simplex. When we are working with the replicator equation 

60 it is straightforward to calculate all the gradients on the unit simplex. However, when working 

61 with finite populations using the social learning model (social imitation), we are actually working 

62 with a simplex with size equivalent to the population size (so all the dimensions of the simplex must 

63 sum to `Z`) and we only consider discrete (integer) values inside the simplex (the population may 

64 only have integer individuals). Of course this can be translated into frequencies, which gets us 

65 back to the unit simplex, but it is not so simple to transform any value between 0-1 sampled with 

66 numpy.linspace to a discrete value. 

67 

68 Therefore, for the discrete case, will will sample directly discrete points in barycentric 

69 coordinates and only then, translate them into cartesian cooordinates. 

70 

71 Parameters 

72 ---------- 

73 nb_points : int 

74 number of points for which to calculate the gradients 

75 discrete : bool 

76 indicates whether we are in the continuous or discrete case 

77 size : int 

78 if we are in the discrete case, indicates the size of the simplex 

79 

80 See Also 

81 -------- 

82 egttools.plotting.plot_gradient, 

83 egttools.plotting.draw_invasion_diagram, 

84 egttools.analytical.replicator_equation, 

85 egttools.analytical.StochDynamics 

86 

87 Cite 

88 ----- 

89 This class has been inspired from: https://github.com/marvinboe/egtsimplex

253 def apply_simplex_boundaries_to_gradients(self, u: np.ndarray, v: np.ndarray) -> SelfSimplex2D: 

254 """ 

255 Applies boundaries of the triangle to a list of gradient values over the cartesian grid. 

256 

257 The boundaries are applied using the X Y grid defined in the instantiation of the class. 

258 

259 Parameters 

260 ---------- 

261 u: numpy.ndarray 

262 The X component of the gradients. 

263 v: numpy.ndarray 

264 The Y component of the gradients 

265 Returns 

266 ------- 

267 Simplex2D 

268 A reference to the class object. 

269 """ 

270 self.Ux = u.copy() 

271 self.Uy = v.copy() 

272 

273 self.Ux = np.where(self.X >= self.Y / self.side_slope, self.Ux, np.nan) 

274 self.Ux = np.where(self.Y <= -self.side_slope * self.X + self.side_slope, self.Ux, np.nan) 

275 self.Ux = np.where(self.Y <= self.top_corner, self.Ux, np.nan) 

276 self.Ux = np.where(self.Y > 0, self.Ux, np.nan) 

277 self.Uy = np.where(self.X >= self.Y / self.side_slope, self.Uy, np.nan) 

278 self.Uy = np.where(self.Y <= -self.side_slope * self.X + self.side_slope, self.Uy, np.nan) 

279 self.Uy = np.where(self.Y <= self.top_corner, self.Uy, np.nan) 

280 self.Uy = np.where(self.Y > 0, self.Uy, np.nan) 

281 

282 return self

284 def draw_triangle(self, color: Optional[str] = 'k', linewidth: Optional[int] = 2, 

285 linewidth_random_drift: Optional[int] = 4, 

286 offset: Optional[float] = 0.0, 

287 corner_gap: Optional[float] = 0.05) -> SelfSimplex2D: 

288 """ 

289 Draws the borders of a triangle enclosing the 2-simplex. 

290 

291 When ``offset > 0`` the three edges are drawn as separate line segments, each shifted 

292 outward (away from the centroid) by ``offset`` units. The segments are shortened at 

293 both ends by ``corner_gap`` (in the edge's own direction) so they do not meet at the 

294 corners, giving a floating-frame appearance that works well when the interior is filled 

295 with discrete markers. 

296 

297 Parameters 

298 ---------- 

299 color: Optional[str] 

300 The color of the borders of the triangle. 

301 linewidth: Optional[int] 

302 The width of the borders of the triangle. 

303 linewidth_random_drift: Optional[int] 

304 The width of the dashed line that represents the edges with random drift. 

305 offset: Optional[float] 

306 Distance to push each edge outward from the centroid. 0 (default) gives the 

307 original behaviour with edges meeting at corners. 

308 corner_gap: Optional[float] 

309 When offset > 0, fraction of the edge length to leave undrawn at each end so the 

310 segments appear visually separated at corners. Has no effect when offset == 0. 

311 

312 Returns 

313 ------- 

314 Simplex2D 

315 A reference to the class object. 

316 """ 

317 if offset == 0.0: 

318 self.ax.triplot(self.triangle, color=color, linewidth=linewidth) 

319 else: 

320 centroid = self.corners.mean(axis=0) 

321 # edges as pairs of corner indices: bottom, right, left 

322 edges = [(0, 2), (2, 1), (1, 0)] 

323 for i, j in edges: 

324 p0, p1 = self.corners[i].copy(), self.corners[j].copy() 

325 edge_vec = p1 - p0 

326 edge_len = np.linalg.norm(edge_vec) 

327 edge_dir = edge_vec / edge_len 

328 

329 # outward normal: perpendicular to edge, pointing away from centroid 

330 normal = np.array([-edge_dir[1], edge_dir[0]]) 

331 mid = (p0 + p1) / 2 

332 if np.dot(normal, mid - centroid) < 0: 

333 normal = -normal 

334 

335 # shift edge outward 

336 p0_off = p0 + offset * normal 

337 p1_off = p1 + offset * normal 

338 

339 # pull endpoints inward along edge to leave a gap at corners 

340 gap = corner_gap * edge_len 

341 p0_off = p0_off + gap * edge_dir 

342 p1_off = p1_off - gap * edge_dir 

343 

344 lw = linewidth_random_drift if any( 

345 (i in e and j in e) for e in self.random_drift_edges 

346 ) else linewidth 

347 ls = 'dashed' if any( 

348 (i in e and j in e) for e in self.random_drift_edges 

349 ) else 'solid' 

350 self.ax.plot([p0_off[0], p1_off[0]], [p0_off[1], p1_off[1]], 

351 color=color, linewidth=lw, linestyle=ls) 

352 

353 if offset == 0.0: 

354 for edge in self.random_drift_edges: 

355 self.ax.plot([self.corners[edge[0], 0], self.corners[edge[1], 0]], 

356 [self.corners[edge[0], 1], self.corners[edge[1], 1]], 

357 lw=linewidth_random_drift, linestyle='dashed', color=color) 

358 

359 return self

454 def draw_stationary_points(self, roots: List[Union[Tuple[float, float], np.ndarray]], 

455 stability: Union[List[bool], List[int]], 

456 zorder: Optional[int] = 5, linewidth: Optional[float] = 3, 

457 atol: Optional[float] = 1e-7) -> SelfSimplex2D: 

458 """ 

459 Draws the black circles for stable points and white circles for unstable ones. 

460 

461 Parameters 

462 ---------- 

463 roots: 

464 A list of arrays (or tuples) containing the cartesian coordinates of the roots. 

465 stability: 

466 A list of boolean or integer values indicating whether the root is stable. If there are integer values 

467 -1 - unstable, 0 - saddle, 1 - stable. 

468 zorder: 

469 Indicates in which order these points should appear in the figure (above or below other plots). 

470 linewidth: 

471 Width of the border of the circles that represents the roots. 

472 atol: 

473 Tolerance to consider a value equal to 0. Used to check if a point is on an edge. 

474 

475 Returns 

476 ------- 

477 Simplex2D 

478 A reference to the class object. 

479 

480 """ 

481 for i, stationary_point in enumerate(roots): 

482 point = xy_to_barycentric_coordinates(stationary_point[0], stationary_point[1], self.corners) 

483 place, _ = find_where_point_is_in_simplex(point) 

484 if place == 1: 

485 # First let's check if stationary point is in an edge with random drift 

486 if np.isclose([point[3 - np.sum(edge)] for edge in self.random_drift_edges], 0., 

487 atol=atol).any(): 

488 continue 

489 

490 if stability[i] == 1: 

491 facecolor = 'k' 

492 elif stability[i] == -1: 

493 facecolor = 'white' 

494 else: 

495 facecolor = 'grey' 

496 self.ax.add_artist(Circle(stationary_point, 0.015, 

497 edgecolor='k', facecolor=facecolor, zorder=zorder, linewidth=linewidth)) 

498 return self

536 def _draw_path(self, v: np.ndarray, color, cmap, linewidth, zorder, draw_arrow: bool = False, 

537 arrowsize=None, arrowstyle=None, position=None, arrowdirection='right'): 

538 """Draws one trajectory (xy points), in a single colour or coloured along time by ``cmap``.""" 

539 if cmap is None: 

540 line = self.ax.plot(v[:, 0], v[:, 1], color, linewidth=linewidth, zorder=zorder)[0] 

541 arrow_color = None 

542 else: 

543 colormap = plt.get_cmap(cmap) if isinstance(cmap, str) else cmap 

544 segments = np.stack([v[:-1], v[1:]], axis=1) 

545 self.ax.add_collection(LineCollection(segments, colors=colormap(np.linspace(0, 1, len(segments))), 

546 linewidths=linewidth, zorder=zorder)) 

547 # Invisible line that carries the geometry for add_arrow. 

548 line = self.ax.plot(v[:, 0], v[:, 1], linewidth=linewidth, alpha=0., zorder=zorder)[0] 

549 where = v[:, 0].mean() if position is None else position 

550 arrow_color = colormap(np.argmin(np.abs(v[:, 0] - where)) / max(len(v) - 1, 1)) 

551 if draw_arrow: 

552 add_arrow(line, size=arrowsize, arrowstyle=arrowstyle, position=position, direction=arrowdirection, 

553 color=arrow_color) 

554 return line

556 def _scatter_path(self, v: np.ndarray, s, color, cmap, marker, zorder): 

557 """Scatters the points of one trajectory, in a single colour or coloured along time by ``cmap``.""" 

558 if cmap is None:

687 def draw_trajectory_from_roots(self, f: Callable[[np.ndarray, int], np.ndarray], roots: List[np.ndarray], 

688 stability: List[bool], 

689 trajectory_length: Optional[int] = 15, step: Optional[float] = 0.1, 

690 perturbation: Optional[Union[int, float]] = 0.01, 

691 color: Optional[Union[str, Tuple[int, int, int]]] = 'k', 

692 linewidth: Optional[float] = 0.5, zorder: Optional[int] = 0, 

693 draw_arrow: Optional[bool] = False, arrowstyle: Optional[str] = 'fancy', 

694 arrowsize: Optional[int] = 50, 

695 position: Optional[int] = None, 

696 arrowdirection: Optional[str] = 'right', 

697 atol: Optional[float] = 1e-7, 

698 cmap: Optional[Union[str, matplotlib.colors.Colormap]] = None) -> SelfSimplex2D: 

699 """ 

700 Draws trajectories inside the unit simplex starting from the stationary points. 

701 

702 Parameters 

703 ---------- 

704 f: Callable[[np.ndarray, int], np.ndarray] 

705 Function that can calculate the gradient at any point in the simplex. 

706 roots: List[np.ndarray[np.float64[3,m]] 

707 A list of points in barycentric coordinates from which the trajectories should start. 

708 stability: List[bool] 

709 Indicates whether the root is a stable or unstable point. 

710 trajectory_length: Optional[int] 

711 Length of the trajectory. This is used to calculate the amount of points odeint should calculate. 

712 step: Optional[float] 

713 The step size in time to get to the maximum trajectory length. Together with trajectory_length 

714 this indicates the amount of points odeint should calculate. 

715 perturbation: Optional[Union[int, float]] 

716 Indicates how much perturbation should be applied to the root to start drawing the trajectory. 

717 If no perturbation is applied, since the gradient is 0, the system will never leave the root. 

718 color: Optional[Union[str, Tuple[int, int, int]]] 

719 The color of the points of the trajectory. 

720 linewidth: Optional[float] = 0.5 

721 Width of the line to be plot. 

722 zorder: Optional[int] 

723 The order in which this plot should appear in the figure (above or bellow other plots). 

724 draw_arrow: Optional[bool] 

725 Indicates whether to draw an arrow along the trajectory. 

726 arrowstyle: Optional[str] 

727 Indicates the style of the arrow to be plotted. 

728 arrowsize: Optional[int] 

729 The size of the arrow. 

730 position: Optional[int] 

731 Where should the arrow be pltoted. 

732 arrowdirection: Optional[str] 

733 Indicates whether the arrow should be plotted in the direction of the advancing trajectory (right) or 

734 the opposite. 

735 atol: Optional[float] 

736 Tolerance to consider a value equal to 0. Used to check if a point is on an edge of the simplex. 

737 

738 cmap: Optional[Union[str, matplotlib.colors.Colormap]] 

739 If given, colour each trajectory along time with this colormap (start to end) instead 

740 of a single ``color``; arrows take the colour of the trajectory where they are drawn. 

741 

742 Returns 

743 ------- 

744 Simplex2D 

745 A reference to the current object. 

746 """ 

747 if self.discrete: 

748 if type(perturbation) is float: 

749 perturbation = 1 

750 

751 for i, stationary_point in enumerate(roots): 

752 # First let's check if stationary point is in an edge with random drift 

753 if np.isclose([stationary_point[3 - np.sum(edge)] for edge in self.random_drift_edges], 0., 

754 atol=atol).any(): 

755 continue 

756 if stability[i]: # we don't plot arrows starting at stable points 

757 continue 

758 stationary_point_discrete = (stationary_point * self.size) 

759 states = perturb_state_discrete(stationary_point_discrete, self.size, perturbation=perturbation) 

760 for state in states: 

761 x = odeint(f, state, np.arange(0, trajectory_length, step), full_output=False) 

762 # check if point is outside simplex 

763 # noinspection PyUnresolvedReferences 

764 tmp_check = np.where(~np.isclose(x.sum(axis=1) / self.size, 1., atol=1e-2))[0] 

765 if len(tmp_check) > 0: 

766 last_inside = np.min(tmp_check) 

767 if last_inside == 0: 

768 continue 

769 x = x[:last_inside] 

770 # noinspection PyTypeChecker 

771 v = barycentric_to_xy_coordinates(x / self.size, self.corners) 

772 self._draw_path(v, color, cmap, linewidth, zorder, draw_arrow, arrowsize, arrowstyle, 

773 position, arrowdirection) 

774 else: 

775 for i, stationary_point in enumerate(roots): 

776 # First let's check if stationary point is in an edge with random drift 

777 if np.isclose([stationary_point[3 - np.sum(edge)] for edge in self.random_drift_edges], 0., 

778 atol=atol).any(): 

779 continue 

780 if stability[i] == 1: # we don't plot arrows starting at stable points 

781 continue 

782 states = perturb_state(stationary_point, perturbation=perturbation) 

783 for state in states: 

784 x = odeint(f, state, np.arange(0, trajectory_length, step), full_output=False) 

785 

786 # noinspection PyTypeChecker 

787 v = barycentric_to_xy_coordinates(x, self.corners) 

788 self._draw_path(v, color, cmap, linewidth, zorder, draw_arrow, arrowsize, arrowstyle, 

789 position, arrowdirection) 

790 

791 return self

886 def draw_stationary_distribution_discrete( 

887 self, 

888 stationary_distribution: np.ndarray, 

889 marker: Optional[str] = 'h', 

890 marker_size: Optional[float] = None, 

891 cmap: Optional[Union[str, matplotlib.colors.Colormap]] = 'binary', 

892 alpha: Optional[float] = 1., 

893 vmin: Optional[float] = None, 

894 vmax: Optional[float] = None, 

895 zorder: Optional[int] = 1, 

896 colorbar: Optional[bool] = True, 

897 aspect: Optional[float] = 10, 

898 anchor: Optional[Tuple[float, float]] = (-0.5, 0.5), 

899 panchor: Optional[Tuple[float, float]] = (0, 0), 

900 shrink: Optional[float] = 0.6, 

901 label: Optional[str] = 'stationary distribution', 

902 label_rotation: Optional[int] = 270, 

903 label_fontsize: Optional[int] = 16, 

904 labelpad: Optional[float] = 20) -> SelfSimplex2D: 

905 """ 

906 Draws the stationary distribution as uniformly-sized discrete markers placed at 

907 each population state on the simplex, coloured by probability. 

908 

909 All markers are the same size; the stationary probability is encoded purely through 

910 colour (using ``cmap``). This makes the tiling nature of the discrete simplex explicit 

911 while avoiding the perceptual ambiguity of size-encoded probability. 

912 

913 Parameters 

914 ---------- 

915 stationary_distribution: numpy.ndarray 

916 Array of stationary probabilities, one per state, in the order produced by 

917 :func:`egttools.sample_simplex` when iterating from 0 to nb_states. 

918 marker: Optional[str] 

919 Matplotlib marker string. Use ``'h'`` for hexagons (default) or ``'o'`` for 

920 circles. Any valid matplotlib marker is accepted. 

921 marker_size: Optional[float] 

922 Size of every marker in points². If ``None`` (default) it is computed 

923 automatically so that markers roughly tile the simplex without overlap, 

924 based on the population size and the current figure dimensions. 

925 cmap: Optional[Union[str, matplotlib.colors.Colormap]] 

926 Colormap used to colour the markers by probability value. 

927 alpha: Optional[float] 

928 Marker transparency. 

929 vmin: Optional[float] 

930 Lower bound of the colormap range. Defaults to the minimum of the distribution. 

931 vmax: Optional[float] 

932 Upper bound of the colormap range. Defaults to the maximum of the distribution. 

933 zorder: Optional[int] 

934 Drawing order relative to other artists on the axes. 

935 colorbar: Optional[bool] 

936 Whether to add a colourbar. 

937 aspect: Optional[float] 

938 Aspect ratio of the colourbar. 

939 anchor: Optional[Tuple[float, float]] 

940 Anchor point of the colourbar axes. 

941 panchor: Optional[Tuple[float, float]] 

942 Parent anchor point for the colourbar. 

943 shrink: Optional[float] 

944 Fraction by which to shrink the colourbar. 

945 label: Optional[str] 

946 Colourbar label. 

947 label_rotation: Optional[int] 

948 Rotation of the colourbar label in degrees. 

949 label_fontsize: Optional[int] 

950 Font size of the colourbar label. 

951 labelpad: Optional[float] 

952 Padding between the colourbar and its label. 

953 

954 Returns 

955 ------- 

956 Simplex2D 

957 A reference to the current object. 

958 """ 

959 if not self.discrete: 

960 raise Exception( 

961 "draw_stationary_distribution_discrete requires a discrete simplex " 

962 "(pass discrete=True and size=Z when constructing Simplex2D)." 

963 ) 

964 

965 # Retrieve the xy position of every discrete state (stored during __init__) 

966 x = self.triangle_discrete.x 

967 y = self.triangle_discrete.y 

968 

969 _vmax = vmax if vmax is not None else stationary_distribution.max() 

970 _vmin = vmin if vmin is not None else stationary_distribution.min() 

971 

972 if marker_size is None: 

973 # Auto-size: fit markers so they tile the simplex without overlapping. 

974 # 

975 # All adjacent discrete states are separated by exactly 1/size in 

976 # Cartesian data units (equilateral triangular lattice). 

977 # 

978 # Scatter `s` is in points² (1 pt = 1/72 inch). We convert the 

979 # inter-state spacing from data units → inches → points using the 

980 # axis position (figure fractions) and the figure size in inches, 

981 # then account for the actual data range visible in each axis. 

982 # Using 72 pt/inch (not figure.dpi, which gives pixels not points). 

983 ax_pos = self.ax.get_position() # fractional position in figure 

984 fig_w_pts = self.figure.get_figwidth() * 72.0 

985 fig_h_pts = self.figure.get_figheight() * 72.0 

986 ax_w_pts = ax_pos.width * fig_w_pts 

987 ax_h_pts = ax_pos.height * fig_h_pts 

988 xlim = self.ax.get_xlim() 

989 ylim = self.ax.get_ylim() 

990 pts_per_x = ax_w_pts / (xlim[1] - xlim[0]) 

991 pts_per_y = ax_h_pts / (ylim[1] - ylim[0]) 

992 # Use the tighter scale so markers never overlap along either axis 

993 spacing_pts = (1.0 / self.size) * min(pts_per_x, pts_per_y) 

994 # Diameter = 0.8 × spacing → 20 % gap between adjacent markers 

995 marker_size = np.pi * (0.8 * spacing_pts / 2) ** 2 

996 

997 sc = self.ax.scatter(x, y, s=marker_size, c=stationary_distribution, 

998 cmap=cmap, alpha=alpha, vmin=_vmin, vmax=_vmax, 

999 marker=marker, zorder=zorder, linewidths=0) 

1000 

1001 if colorbar: 

1002 cbar = self.figure.colorbar(sc, aspect=aspect, anchor=anchor, panchor=panchor, 

1003 shrink=shrink, ax=self.ax) 

1004 cbar.set_label(label, rotation=label_rotation, fontsize=label_fontsize, labelpad=labelpad) 

1005 

1006 return self
